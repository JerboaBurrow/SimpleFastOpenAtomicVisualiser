273 "#version " GLSL_VERSION "\n"
274 "precision lowp float; precision lowp int;\n"
275 "in vec2 billboard;\n"
276 "in vec3 aViewPos;\n"
277 "in vec3 bViewPos;\n"

288 "uniform float ambientLight;\n"
289 "uniform float bondScale;\n"
290 "bool sphereHit(vec3 rayDirection, vec3 centre, float radius, out vec3 pos, out vec3 normal)\n"
291 "{\n"
292 " float b = 2.0 * dot(rayDirection, -centre);\n"
293 " float r2 = radius*radius;\n"
294 " float determinant = b * b - (4.0 * (dot(centre, centre) - r2));\n"
295 " if(determinant < 0.0) { return false; }\n"
296 " determinant = sqrt(determinant);\n"
297 " pos = rayDirection * min((-b+determinant)*0.5, (-b-determinant)*0.5);\n"
298 " normal = normalize(pos - centre);\n"
299 " return true;\n"
300 "}\n"
301 /*
302 * Creates a capsule (sphere integrated along a line).
303 * .-.B
304 * / / \
305 * / / /
306 * / / /
307 * / / /
308 * / / /
309 * \ / /
310 * A--
311 *
312 * To derive, consider in view space a ray r(t) = t*v, and a capsule with midline c(s) = a+s*n.
313 * Then solve the quadratic ||a-r(t)-[(a-r(t).n)]n||^2-R^2 = 0 (=> r(t)->midline distance = R);
314 * Also then project the hit point to the midline for normals.
315 */
316 "bool capsuleHit"
317 "("
318 " vec3 rayDirection,"
319 " vec3 capsuleDirection,"
320 " vec3 capsuleA,"
321 " vec3 capsuleB,"
322 " float radius,"
323 " float midLength,"
324 " out vec3 viewPos,"
325 " out vec3 viewNormal,"
326 " out vec3 projectedHitPoint,"
327 " out float s"
328 ")\n"
329 "{\n"
330 " float a = dot(capsuleA, capsuleDirection);"
331 " float b = dot(rayDirection, capsuleDirection);\n"
332 " float c = dot(capsuleA, rayDirection);\n"
333 " float l = dot(capsuleA, capsuleA);\n"
334 " float r2 = radius*radius;\n"
335 " float determinant = (a*b-c)*(a*b-c)-(1.0-b*b)*(l-a*a-r2);\n"
336 " if (determinant < 0)\n"
337 " {\n"
338 " if (sphereHit(rayDirection, capsuleA, radius, viewPos, viewNormal)){ projectedHitPoint = capsuleA; s = 0.0; return true; }\n"
339 " if (sphereHit(rayDirection, capsuleB, radius, viewPos, viewNormal)){ projectedHitPoint = capsuleB; s = midLength; return true; }\n"
340 " return false;"
341 " \n}"
342 " float d = sqrt(determinant);\n"
343 // Ray hit point.
344 " float t = min( (-(a*b-c)+d)/(1.0-b*b), (-(a*b-c)-d)/(1.0-b*b) );"
345 " viewPos = rayDirection * t;\n"
346 " s = -dot(capsuleA-viewPos, capsuleDirection);\n"
347 " if (s < 0){\n"
348 " if (sphereHit(rayDirection, capsuleA, radius, viewPos, viewNormal)){ projectedHitPoint = capsuleA; s = 0.0; return true; }\n"
349 " return false;"
350 " \n}"
351 " if (s > midLength){\n"
352 " if (sphereHit(rayDirection, capsuleB, radius, viewPos, viewNormal)){ projectedHitPoint = capsuleB; s = midLength; return true; }\n"
353 " return false;"
354 " }"
355 " vec3 u = s*capsuleDirection;\n"
356 " projectedHitPoint = capsuleA+u;\n"
357 " viewNormal = normalize(viewPos-projectedHitPoint);\n"
358 " return true;\n"
359 "}"
360 "void main()\n"
361 "{\n"
362 " vec3 lightViewPos = (view*lightPos).xyz;\n"
363 " vec3 rayDirection = normalize(vec3(billboard * bondScale, 0.0) + comViewPos);"
364 " float midLength = length(bViewPos-aViewPos);\n"
365 " vec3 capsuleDirection = normalize(bViewPos-aViewPos);\n"
366 " vec3 viewPos; vec3 viewNormal; vec3 projectedHitPoint; float s;\n"
367 " bool hit = capsuleHit(rayDirection, capsuleDirection, aViewPos, bViewPos, bondScale, midLength, viewPos, viewNormal, projectedHitPoint, s);\n"
368 " if (!hit) { discard; }\n"
369 " else\n"
370 " {\n"
371 " vec4 clipPos = proj * vec4(viewPos, 1.0);\n"
372 " float ndcDepth = clipPos.z / clipPos.w;\n"
373 " gl_FragDepth = ((gl_DepthRange.diff * ndcDepth) + gl_DepthRange.near + gl_DepthRange.far) / 2.0;\n"
374 " float diff = max(dot(viewNormal, normalize(lightViewPos-projectedHitPoint)), 0.0);\n"
375 " vec4 col = a_colour;\n"
376 " if (s > 0.5*midLength) { col = b_colour; }\n"
377 " colour = vec4((ambientLight + diff)*lightColour.rgb * col.rgb, col.a);\n"
378 " }\n"

34glm::vec3 cartesian2Spherical(glm::vec3 xyz)
35{
36 float r = glm::length(xyz);
37 float sgn = (-1.0*xyz.z) < 0 ? -1.0 : 1.0;
38 return
39 {
40 r,
41 std::acos(xyz.y/r),
42 sgn * std::acos(xyz.x/std::sqrt(xyz.x*xyz.x+xyz.z*xyz.z))
43 };
44}

62glm::vec3 spherical2cartesian(glm::vec3 rthetaphi)
63{
64
65 return
66 {
67 rthetaphi.x*std::sin(rthetaphi.y)*std::cos(rthetaphi.z),
68 rthetaphi.x*std::cos(rthetaphi.y),
69 -rthetaphi.x*std::sin(rthetaphi.y)*std::sin(rthetaphi.z)
70 };
71}

106std::vector<std::string> split(std::string str, std::regex delim)
107{
108 std::vector<std::string> s;
109 auto matches = std::sregex_iterator(str.begin(), str.end(), delim);
110 if (std::distance(matches, std::sregex_iterator()) == 0) { return {str}; }
111 auto key = matches->position();
112 if (key > 0)
113 {
114 s.push_back(str.substr(0, key-1));
115 }
116 while (matches != std::sregex_iterator())
117 {
118 key = matches->position();
119 matches++;
120 auto next = matches->position();
121 s.push_back(str.substr(key, next-1-key));
122 }
123 return s;
124}

162bool atomControls
163(
164 jGL::DesktopDisplay & display,
165 std::vector<Atom> & atoms,
166 std::map<int, Element> & emphasisControls,
167 std::multimap<Element, uint64_t> & elementMap,
168 std::vector<float> & alphaOverrides,
169 float deemphasisAlpha
170)
171{
172 bool elementsNeedUpdate = false;
173 if (display.keyHasEvent(GLFW_KEY_LEFT, jGL::EventType::PRESS) || display.keyHasEvent(GLFW_KEY_LEFT, jGL::EventType::HOLD))
174 {
175 translate(atoms, {-dr, 0.0, 0.0});
176 elementsNeedUpdate = true;
177 }
178 if (display.keyHasEvent(GLFW_KEY_RIGHT, jGL::EventType::PRESS) || display.keyHasEvent(GLFW_KEY_RIGHT, jGL::EventType::HOLD))
179 {
180 translate(atoms, {dr, 0.0, 0.0});
181 elementsNeedUpdate = true;
182 }
183 if (display.keyHasEvent(GLFW_KEY_PERIOD, jGL::EventType::PRESS) || display.keyHasEvent(GLFW_KEY_PERIOD, jGL::EventType::HOLD))
184 {
185 translate(atoms, {0.0, -dr, 0.0});
186 elementsNeedUpdate = true;
187 }
188 if (display.keyHasEvent(GLFW_KEY_SLASH, jGL::EventType::PRESS) || display.keyHasEvent(GLFW_KEY_SLASH, jGL::EventType::HOLD))
189 {
190 translate(atoms, {0.0, dr, 0.0});
191 elementsNeedUpdate = true;
192 }
193 if (display.keyHasEvent(GLFW_KEY_DOWN, jGL::EventType::PRESS) || display.keyHasEvent(GLFW_KEY_DOWN, jGL::EventType::HOLD))
194 {
195 translate(atoms, {0.0, 0.0, -dr});
196 elementsNeedUpdate = true;
197 }
198 if (display.keyHasEvent(GLFW_KEY_UP, jGL::EventType::PRESS) || display.keyHasEvent(GLFW_KEY_UP, jGL::EventType::HOLD))
199 {
200 translate(atoms, {0.0, 0.0, dr});
201 elementsNeedUpdate = true;
202 }
203
204 for (const auto & control : emphasisControls)
205 {
206 if (display.keyHasEvent(control.first, jGL::EventType::PRESS))
207 {
208 auto iter = elementMap.equal_range(control.second);
209 while (iter.first != iter.second)
210 {
211 float & alpha = alphaOverrides[iter.first->second];
212 alpha = (alpha == emphasisedAlpha ? deemphasisAlpha : emphasisedAlpha);
213 atoms[iter.first->second].colour.a = alpha;
214 iter.first++;
215 }
216 elementsNeedUpdate = true;
217 }
218 }
219
220 return elementsNeedUpdate;
221}

242void loadingScreenFrame

246 AtomRenderer & loadingAtoms,
247 std::string progressMessage,
248 Theme theme,
249 const unsigned int resX,
250 const unsigned int resY,

278 }
279
280 jGLInstance->endFrame();
281 display.loop();
282
283 delta = 0.0;
284 for (int n = 0; n < 60; n++)
285 {
286 delta += deltas[n];
287 }
288 delta /= 60.0;
289 auto toc = std::chrono::high_resolution_clock::now();
290 deltas[frameId] = std::chrono::duration_cast<std::chrono::milliseconds>(toc-tic).count();
291 frameId = (frameId+1) % 60;

228 CommandLine(uint8_t count, char ** commandLine)
229 {
230 if (count == 1) { help(); return; }
231 for (uint8_t c = 1; c < count; c++)
232 {
233 checkSpecial(commandLine[c]);
234 }
235 welcome();
236 getArgument<std::filesystem::path>(structure, commandLine[1]);
237
238 std::vector<std::string> flags;
239 std::map<std::string, const char *> arguments;
240
241 for (uint8_t c = 2; c < count; c++)
242 {
243 std::string arg = commandLine[c];
244 if (isFlag(arg))
245 {
246 flags.push_back(stripPrefix(commandLine[c]));
247 if (c+1 < count && isValue(commandLine[c+1]))
248 {
249 std::cout << "Option " << arg << " looks like a flag, it should not have a value.\n";
250 std::exit(EXIT_FAILURE);
251 }
252 }
253 else
254 {
255 if (c+1 < count && isValue(commandLine[c+1]))
256 {
257 arguments[stripPrefix(commandLine[c])] = commandLine[c+1];
258 c += 1;
259 }
260 else
261 {
262 std::cout << "Option " << arg << " is missing its value.\n";
263 std::exit(EXIT_FAILURE);
264 }
265 }

469 inline int lua_getOption(lua_State * lua)
470 {
471 int args = lua_gettop(lua);
472 if (args != 1)
473 {
474 const std::string msg = "setOption expects a name as argument.\n";
475 lua_pushlstring(lua, msg.c_str(), msg.length());
476 return lua_error(lua);
477 }
478 LuaString s; s.read(lua, 1);
479
480 bool * bval = boolFromName(s.characters);
481 if (bval != nullptr)
482 {
483 lua_pushboolean(lua, *bval);
484 return 1;
485 }
486
487 float * fval = floatFromName(s.characters);
488 if (fval != nullptr)
489 {
490 lua_pushnumber(lua, *fval);
491 return 1;
492 }
493
494 uint8_t * i8val = i8FromName(s.characters);
495 if (i8val != nullptr)
496 {
497 lua_pushinteger(lua, *i8val);
498 return 1;
499 }
500
501 uint64_t * i64val = i64FromName(s.characters);
502 if (i64val != nullptr)
503 {
504 lua_pushinteger(lua, *i64val);
505 return 1;
506 }
507
508 std::string * sval = stringFromName(s.characters);
509 if (sval != nullptr)
510 {
511 lua_pushstring(lua, sval->c_str());
512 return 1;
513 }
514 const std::string msg = "getOption, no option " + s.characters + ".\n";
515 lua_pushlstring(lua, msg.c_str(), msg.length());
516 return lua_error(lua);
517 }
